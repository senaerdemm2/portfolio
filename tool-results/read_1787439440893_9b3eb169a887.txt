   159→/*  ANIMATED COUNTER                                                   */
   160→/* ------------------------------------------------------------------ */
   161→
   162→function AnimatedCounter({ target, suffix = "" }: { target: number; suffix?: string }) {
   163→  const [count, setCount] = useState(0);
   164→  const ref = useRef<HTMLSpanElement>(null);
   165→  const inView = useInView(ref, { once: true, margin: "-50px" });
   166→
   167→  useEffect(() => {
   168→    if (!inView) return;
   169→    let start = 0;
   170→    const duration = 2000;
   171→    const step = (timestamp: number) => {
   172→      if (!start) start = timestamp;
   173→      const progress = Math.min((timestamp - start) / duration, 1);
   174→      const eased = 1 - Math.pow(1 - progress, 3);
   175→      setCount(Math.floor(eased * target));
   176→      if (progress < 1) requestAnimationFrame(step);
   177→    };
   178→    requestAnimationFrame(step);
   179→  }, [inView, target]);
   180→
   181→  return (
   182→    <span ref={ref} className="tabular-nums">
   183→      {count}
   184→      {suffix}
   185→    </span>
   186→  );
   187→}
