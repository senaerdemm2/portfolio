   191→/*  ANIMATED COUNTER                                                   */
   192→/* ------------------------------------------------------------------ */
   193→
   194→function AnimatedCounter({ target, suffix = "" }: { target: number; suffix?: string }) {
   195→  const [count, setCount] = useState(0);
   196→  const ref = useRef<HTMLSpanElement>(null);
   197→  const inView = useInView(ref, { once: true, margin: "-50px" });
   198→
   199→  useEffect(() => {
   200→    if (!inView) return;
   201→    let start = 0;
   202→    const duration = 2000;
   203→    const step = (timestamp: number) => {
   204→      if (!start) start = timestamp;
   205→      const progress = Math.min((timestamp - start) / duration, 1);
   206→      const eased = 1 - Math.pow(1 - progress, 3);
   207→      setCount(Math.floor(eased * target));
   208→      if (progress < 1) requestAnimationFrame(step);
   209→    };
   210→    requestAnimationFrame(step);
   211→  }, [inView, target]);
   212→
   213→  return (
   214→    <span ref={ref} className="tabular-nums">
   215→      {count}
   216→      {suffix}
   217→    </span>
   218→  );
   219→}

   222→/*  DASHBOARD IMAGE CAROUSEL                                           */
   223→/* ------------------------------------------------------------------ */
   224→
   225→function DashboardCarousel({ images }: { images: string[] }) {
   226→  const [current, setCurrent] = useState(0);
   227→  const [lightbox, setLightbox] = useState(false);
   228→
   229→  const next = useCallback(() => setCurrent((i) => (i + 1) % images.length), [images.length]);
   230→  const prev = useCallback(() => setCurrent((i) => (i - 1 + images.length) % images.length), [images.length]);
   231→
   232→  if (!images || images.length === 0) return null;
   233→
   234→  return (
   235→    <>
   236→      {/* Thumbnail strip */}
   237→      <div className="mt-5 relative">
   238→        <div className="flex gap-3 overflow-x-auto pb-2 scrollbar-hide">
   239→          {images.map((img, i) => (
   240→            <button
   241→              key={img}
   242→              onClick={() => { setCurrent(i); setLightbox(true); }}
   243→              className={`shrink-0 rounded-lg overflow-hidden border-2 transition-all duration-200 hover:border-primary/50 ${
   244→                i === current && !lightbox ? "border-primary" : "border-border/50"
   245→              }`}
   246→            >
   247→              <img
   248→                src={img}
   249→                alt={`Dashboard ${i + 1}`}
   250→                className="w-48 h-28 sm:w-56 sm:h-32 object-cover object-top"
   251→              />
   252→            </button>
   253→          ))}
   254→        </div>
   255→        <p className="text-[11px] text-muted-foreground/60 mt-2">
   256→          Click to enlarge
   257→        </p>
   258→      </div>
   259→
   260→      {/* Lightbox */}
   261→      <AnimatePresence>
   262→        {lightbox && images.length > 0 && (
   263→          <motion.div
   264→            initial={{ opacity: 0 }}
   265→            animate={{ opacity: 1 }}
   266→            exit={{ opacity: 0 }}
   267→            className="fixed inset-0 z-[100] bg-black/90 backdrop-blur-sm flex items-center justify-center p-4"
   268→            onClick={() => setLightbox(false)}
   269→          >
   270→            {/* Close button */}
   271→            <button
   272→              className="absolute top-4 right-4 text-white/70 hover:text-white z-10"
   273→              onClick={() => setLightbox(false)}
   274→            >
   275→              <XCircle className="h-8 w-8" />
   276→            </button>
   277→
   278→            {/* Nav arrows */}
   279→            {images.length > 1 && (
   280→              <>
   281→                <button
   282→                  className="absolute left-4 top-1/2 -translate-y-1/2 text-white/60 hover:text-white z-10"
   283→                  onClick={(e) => { e.stopPropagation(); prev(); }}
   284→                >
   285→                  <ChevronLeft className="h-10 w-10" />
   286→                </button>
   287→                <button
   288→                  className="absolute right-4 top-1/2 -translate-y-1/2 text-white/60 hover:text-white z-10"
   289→                  onClick={(e) => { e.stopPropagation(); next(); }}
   290→                >
   291→                  <ChevronRight className="h-10 w-10" />
   292→                </button>
   293→              </>
   294→            )}
   295→
   296→            {/* Image counter */}
   297→            <div className="absolute bottom-4 left-1/2 -translate-x-1/2 text-white/50 text-sm">
   298→              {current + 1} / {images.length}
   299→            </div>
   300→
   301→            {/* Image */}
   302→            <motion.div
   303→              key={current}
   304→              initial={{ opacity: 0, scale: 0.95 }}
   305→              animate={{ opacity: 1, scale: 1 }}
   306→              exit={{ opacity: 0, scale: 0.95 }}
   307→              transition={{ duration: 0.2 }}
   308→              className="max-w-[90vw] max-h-[85vh]"
   309→              onClick={(e) => e.stopPropagation()}
   310→            >
   311→              <img
   312→                src={images[current]}
   313→                alt={`Dashboard ${current + 1}`}
   314→                className="max-w-full max-h-[85vh] object-contain rounded-lg"
   315→              />
   316→            </motion.div>
   317→          </motion.div>
   318→        )}
   319→      </AnimatePresence>
   320→    </>
   321→  );
   322→}
